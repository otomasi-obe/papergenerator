7. SCRIPT BATCH VALIDASI (batch_audit.sh)
================================================================================

#!/bin/bash
# Jalankan dari folder Journal/

for gen in *gen.py; do
    name="${gen%gen.py}"
    template="${name}.docx"
    output="${name}_output.docx"
    
    if [ -f "$template" ]; then
        echo "=== Validating $name ==="
        python "$gen" 2>&1 | tail -3
        python -autoDocx.py "$template" "$output" 2>&1 | tail -5
        echo ""
    fi
done
