- Title
- Author
- Affiliation
- AbstractHeading
- Abstract
- Keywords
- Heading1
- Heading2
- BodyText
- Caption
- TableHead
- TableColHead
- TableText
- Reference
- Equation

You can create these by:
1. Copying IEEE.docx as a base
2. Modifying styles to match the target format
3. Or downloading official templates from publisher websites

STEP 7: Frontend Integration
-----------------------------
Update the frontend to use the new template API.

In /home/sirobo/papergenerator/frontend/src/stores/paper.ts:

```typescript
// Add function to fetch templates
export async function fetchTemplates() {
  const response = await axios.get('/api/templates')
  return response.data.templates
}

// Add function to get template details
export async function getTemplateDetails(code: string) {
  const response = await axios.get(`/api/templates/${code}`)
  return response.data
}

// Add function to validate paper
export async function validatePaper(code: string, paperData: any) {
  const response = await axios.post(`/api/templates/${code}/validate`, paperData)
  return response.data
}
```

Create a template selection component:

```vue
<template>
  <div class="template-selector">
    <h3>Select Journal Template</h3>
    <input v-model="searchQuery" placeholder="Search templates..." />
    
    <div class="filters">
      <select v-model="selectedField">
        <option value="">All Fields</option>
        <option v-for="field in fields" :key="field" :value="field">
          {{ field }}
        </option>
      </select>
      
      <select v-model="selectedStyle">
        <option value="">All Citation Styles</option>
        <option v-for="style in citationStyles" :key="style" :value="style">
          {{ style }}
        </option>
      </select>
    </div>
    
    <div class="template-list">
      <div v-for="template in filteredTemplates" 
           :key="template.code"
           @click="selectTemplate(template)"
           :class="{ selected: selectedTemplate?.code === template.code }">
        <h4>{{ template.name }}</h4>
        <p>{{ template.full_name }}</p>
        <div class="meta">
          <span>{{ template.type }}</span>
          <span>{{ template.citation_style }}</span>
          <span>{{ template.columns }} column(s)</span>
        </div>
      </div>
    </div>
  </div>
</template>

<script setup>
import { ref, computed, onMounted } from 'vue'
import { fetchTemplates } from '@/stores/paper'

const templates = ref([])
const searchQuery = ref('')
const selectedField = ref('')
const selectedStyle = ref('')
const selectedTemplate = ref(null)

onMounted(async () => {
  templates.value = await fetchTemplates()
})

const filteredTemplates = computed(() => {
  return templates.value.filter(t => {
    const matchesSearch = !searchQuery.value || 
      t.name.toLowerCase().includes(searchQuery.value.toLowerCase()) ||
      t.full_name.toLowerCase().includes(searchQuery.value.toLowerCase())
    
    const matchesField = !selectedField.value || 
      t.fields.includes(selectedField.value)
    
    const matchesStyle = !selectedStyle.value || 
      t.citation_style === selectedStyle.value
    
    return matchesSearch && matchesField && matchesStyle
  })
})

function selectTemplate(template) {
  selectedTemplate.value = template
  // Emit event or update store
}
</script>
```
